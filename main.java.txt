<div class="container">
  <h1>My ToDo List</h1>
  <p id="progress">0 / 0 完了</p>

  <div id="tasks"></div>

  <input id="newTask" placeholder="タスクを入力">
  <select id="category">
    <option>仕事・用事</option>
    <option>暮らし</option>
    <option>明日の準備</option>
  </select>
  <button onclick="addTask()">＋ 追加</button>
</div>

<script>
  let tasks = [
    { text: "今日の予定を確認", category: "仕事・用事", done: false },
    { text: "優先タスクを終わらせる", category: "仕事・用事", done: false },
    { text: "部屋を片付ける", category: "暮らし", done: false },
    { text: "買い物をする", category: "暮らし", done: false },
    { text: "明日の準備", category: "明日の準備", done: false }
  ];

  function renderTasks() {
    const container = document.getElementById("tasks");
    container.innerHTML = "";

    ["仕事・用事", "暮らし", "明日の準備"]
      .forEach(category => {
        const group = tasks
          .map((task, index) => ({ ...task, index }))
          .filter(task => task.category === category);

        if (group.length === 0) return;

        const heading = document.createElement("h3");
        heading.textContent = category;
        container.appendChild(heading);

        group.forEach(task => {
          const div = document.createElement("div");
          div.className = "task";

          const checkbox = document.createElement("input");
          checkbox.type = "checkbox";
          checkbox.checked = task.done;
          checkbox.onchange = () => {
            tasks[task.index].done = checkbox.checked;
            renderTasks();
          };

          const span = document.createElement("span");
          span.textContent = task.text;
          if (task.done) span.className = "done";

          const del = document.createElement("button");
          del.textContent = "削除";
          del.className = "delete";
          del.onclick = () => {
            tasks.splice(task.index, 1);
            renderTasks();
          };

          div.append(checkbox, span, del);
          container.appendChild(div);
        });
      });

    const completed = tasks.filter(t => t.done).length;
    document.getElementById("progress").textContent =
      `${completed} / ${tasks.length} 完了`;
  }

  function addTask() {
    const input = document.getElementById("newTask");
    const category = document.getElementById("category").value;

    if (!input.value.trim()) return;

    tasks.push({
      text: input.value.trim(),
      category: category,
      done: false
    });

    input.value = "";
    renderTasks();
  }

  renderTasks();
</script>
